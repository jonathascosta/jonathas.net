UPDATE dbo.Customers
SET PostalCode = REGEXP_REPLACE(PostalCode, '^(\d{4})\s*-?\s*(\d{3})$', '\1-\2')
WHERE REGEXP_LIKE(PostalCode, '^\d{4}\s*-?\s*\d{3}$');
